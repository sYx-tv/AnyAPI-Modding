# Give each shipped DLL a readable Windows version and product identity.
function(anyapi_version_resource target product)
  set(ANYAPI_PRODUCT_NAME "${product}")
  set(ANYAPI_VERSION_COMMA "0,27,0,0")
  set(ANYAPI_VERSION_TEXT "0.27.0")
  if(target STREQUAL "dinput8" OR target STREQUAL "AnyGraphics")
    set(ANYAPI_VERSION_COMMA "0,29,0,0")
    set(ANYAPI_VERSION_TEXT "0.29.0")
  endif()
  set(ANYAPI_BINARY_NAME "${product}.dll")
  if(target STREQUAL "dinput8")
    set(ANYAPI_BINARY_NAME "dinput8.dll")
  endif()
  if(target STREQUAL "dinput8")
    set(ANYAPI_VERSION_COMMA "0,34,0,0")
    set(ANYAPI_VERSION_TEXT "0.34.0")
  elseif(target STREQUAL "AnyGraphics")
    set(ANYAPI_VERSION_COMMA "0,29,3,0")
    set(ANYAPI_VERSION_TEXT "0.29.3")
  elseif(target STREQUAL "AnyMapMod")
    set(ANYAPI_VERSION_COMMA "0,27,2,0")
    set(ANYAPI_VERSION_TEXT "0.27.2")
  elseif(target STREQUAL "AnyQuickWheel")
    set(ANYAPI_VERSION_COMMA "1,1,0,0")
    set(ANYAPI_VERSION_TEXT "1.1.0")
  elseif(target STREQUAL "EngineSound")
    set(ANYAPI_VERSION_COMMA "0,1,0,0")
    set(ANYAPI_VERSION_TEXT "0.1.0")
  elseif(target STREQUAL "AnyBalance")
    set(ANYAPI_VERSION_COMMA "1,1,0,0")
    set(ANYAPI_VERSION_TEXT "1.1.0")
  elseif(target STREQUAL "AnyClock")
    set(ANYAPI_VERSION_COMMA "1,0,0,0")
    set(ANYAPI_VERSION_TEXT "1.0.0")
  endif()
  configure_file(version.rc.in "${CMAKE_CURRENT_BINARY_DIR}/${target}-version.rc" @ONLY)
  target_sources(${target} PRIVATE "${CMAKE_CURRENT_BINARY_DIR}/${target}-version.rc")
endfunction()
anyapi_version_resource(dinput8 AnyAPI)
anyapi_version_resource(AnyHelpers AnyHelpers)
anyapi_version_resource(AnyInventory AnyInventory)
anyapi_version_resource(AnyStorage AnyStorage)
anyapi_version_resource(AnyMapMod AnyMap)
anyapi_version_resource(AnyGraphics AnyGraphics)
